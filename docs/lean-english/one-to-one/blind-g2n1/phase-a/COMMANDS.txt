'5-spawn_attenuates.lean': '''import ControlStack.Scenarios.SC09Priv

namespace ControlStack.SC09

-- NOTE
-- AMBIGUOUS: \"sub-collection\" could mean membership inclusion, multiset
-- inclusion (counting repeated capabilities), or an order-preserving
-- sublist. This statement uses membership inclusion: every capability
-- held by a newly appearing process was held by the parent.
def Back_spawn_attenuates : Prop :=
  ∀ (P : Policy) (s : St) (p ch : ℕ) (q : Proc) (keep : List ℕ),
    s.halted = false →
    procOf s p = some q →
    procOf s ch = none →
    ∀ r ∈ (step P full s (.spawn p ch keep)).procs,
      r ∉ s.procs →
      r.lin = q.lin ∧ (∀ c ∈ r.caps, c ∈ q.caps)

end ControlStack.SC09
''',

/bin/bash -lc "python3 - <<'PY'
import json
import subprocess
from datetime import datetime, timezone
from pathlib import Path

result = subprocess.run('sha256sum back/*.lean', shell=True, check=True, capture_output=True, text=True)
print(result.stdout, end='')
hashes = {}
for line in result.stdout.splitlines():
    digest, filename = line.split(maxsplit=1)
    hashes[filename] = digest
manifest = {'utc': datetime.now(timezone.utc).isoformat(timespec='seconds').replace('+00:00', 'Z'), 'files': hashes}
Path('back/BACK-HASHES.json').write_text(json.dumps(manifest, indent=2) + '\\n')
print('Wrote back/BACK-HASHES.json. No Lean files changed after hashing.')
PY"
