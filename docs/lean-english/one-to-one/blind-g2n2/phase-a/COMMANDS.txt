/bin/bash -c "python3 - <<'PY'
import json
import subprocess
from datetime import datetime, timezone
from pathlib import Path

result = subprocess.run('sha256sum back/*.lean', shell=True, check=True,
                        capture_output=True, text=True)
files = {}
for line in result.stdout.splitlines():
    digest, filename = line.split(maxsplit=1)
    files[filename] = digest
utc = datetime.now(timezone.utc).isoformat(timespec='seconds').replace('+00:00', 'Z')
manifest = {'utc': utc, 'files': files}
Path('back/BACK-HASHES.json').write_text(
    json.dumps(manifest, indent=2) + '\\n', encoding='utf-8')
print(result.stdout, end='')
print('Wrote back/BACK-HASHES.json at ' + utc)
PY"
