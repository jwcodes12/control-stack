from pathlib import Path
import hashlib,json,subprocess,datetime,concurrent.futures
out=Path('docs/lean-english/one-to-one/blind-g3n1')
for d in ['typed','compile-fixes','compile-logs']: (out/d).mkdir(exist_ok=True)
expected=json.loads((out/'back/BACK-HASHES.json').read_text())['files']
files=sorted((out/'back').glob('*.lean'),key=lambda p:int(p.name.split('-')[0]))
def compile(p):
 r=subprocess.run(['lake','env','lean',str(p)],capture_output=True,text=True)
 n=p.name.split('-')[0]
 (out/'compile-logs'/f'{n}.original.txt').write_text(r.stdout+r.stderr)
 return p,r.returncode,r.stdout+r.stderr
rows=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
 for p,rc,log in pool.map(compile,files):
  sha=hashlib.sha256(p.read_bytes()).hexdigest()
  assert sha==expected['back/'+p.name],p
  rows.append({'claim':int(p.name.split('-')[0]),'file':str(p),'sha256':sha,'original_compile_exit':rc})
  print(p.name,rc,log[:1500],flush=True)
  if rc==0:
   (out/'typed'/p.name).write_bytes(p.read_bytes())
   (out/'compile-fixes'/f'{p.name.split(\"-\")[0]}.diff').write_text('No repair. Meaning unchanged; typed file is byte-identical to back file.\\n')
(out/'compile-logs/summary.json').write_text(json.dumps(rows,indent=2)+'\\n')
(out/'typed/HASHES.json').write_text(json.dumps({'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'original_declarations_opened':False,'files':{'typed/'+p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in (out/'typed').glob('*.lean')}},indent=2)+'\\n')
PY"
