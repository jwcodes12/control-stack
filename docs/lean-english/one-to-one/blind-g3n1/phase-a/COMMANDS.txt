/bin/bash -lc "python3 - <<'PY'
import json
import subprocess
from datetime import datetime, timezone
from pathlib import Path

result = subprocess.run('sha256sum back/*.lean', shell=True, check=True, text=True, capture_output=True)
print(result.stdout, end='')
files = {}
for line in result.stdout.splitlines():
    digest, filename = line.split(maxsplit=1)
    files[filename] = digest
manifest = {
    'utc': datetime.now(timezone.utc).isoformat(timespec='seconds').replace('+00:00', 'Z'),
    'files': files,
}
Path('back/BACK-HASHES.json').write_text(json.dumps(manifest, indent=2) + '\\n')
print('Wrote back/BACK-HASHES.json; no Lean files changed after hashing.')
PY"
