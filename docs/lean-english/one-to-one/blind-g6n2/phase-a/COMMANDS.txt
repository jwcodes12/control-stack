-- The nonce and haltCheck flags are unrestricted by Sound.
def Back_safe_of_sound : Prop :=
  ∀ (R : Roles) (cap : ℕ) (C : Checks) (ops : List Op),
    Sound C →
    (∀ o ∈ ops, legal R o) →
    Good R cap (run R cap C init ops)

  ∀ (R : Roles) (cap : ℕ) (ops : List Op),

    (∀ o ∈ ops, legal R o) →
    let s := run R cap full init ops
    ∀ e ∈ s.bank,
      ∃ (r : Req) (a : ℕ),

/bin/bash -lc "python3 - <<'PY'
import json
import subprocess
from datetime import datetime, timezone
from pathlib import Path

result = subprocess.run('sha256sum back/*.lean', shell=True, check=True, capture_output=True, text=True)
files = {}
for line in result.stdout.splitlines():
    digest, filename = line.split(maxsplit=1)
    files[filename] = digest
manifest = {
    'utc': datetime.now(timezone.utc).isoformat(timespec='seconds').replace('+00:00', 'Z'),
    'files': files,
}
Path('back/BACK-HASHES.json').write_text(json.dumps(manifest, indent=2) + '\\n')
print(result.stdout, end='')
print('Wrote back/BACK-HASHES.json at ' + manifest['utc'])
PY"
