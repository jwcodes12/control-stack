/bin/bash -lc "python3 - <<'PY'
import subprocess
import json
from datetime import datetime, timezone
from pathlib import Path
result = subprocess.run(['bash', '-c', 'sha256sum back/*.lean'], check=True, capture_output=True, text=True)
print(result.stdout, end='')
files = {}
for line in result.stdout.splitlines():
    digest, name = line.split(maxsplit=1)
    files[name] = digest
manifest = {'utc': datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z'), 'files': files}
Path('back/BACK-HASHES.json').write_text(json.dumps(manifest, indent=2) + '\\n')
print('Wrote back/BACK-HASHES.json at ' + manifest['utc'])
PY"
