def ControlStack.TrustedBroker.Safe : ControlStack.TrustedBroker.State → Prop :=
fun s =>
  And (@LE.le Nat instLENat s.spent s.cap)
    (And (@LE.le Nat instLENat s.releases s.spent)
      (And (@List.Nodup Nat s.consumed)
        (∀ (k : Nat) (l : ControlStack.TrustedBroker.Lease),
          @Eq (Option ControlStack.TrustedBroker.Lease) (s.leases k) (@some ControlStack.TrustedBroker.Lease l) →
            @LE.le Nat instLENat l.used l.budget)))
def ControlStack.TrustedBroker.Coherent : ControlStack.TrustedBroker.State → Prop :=
fun s =>
  ∀ (nonce : Nat),
    Iff (@Membership.mem Nat (List Nat) (@List.instMembership Nat) s.consumed nonce)
      (@Exists ControlStack.TrustedBroker.Approval fun a =>
        And
          (@Eq (Option ControlStack.TrustedBroker.Approval) (s.approvals nonce)
            (@some ControlStack.TrustedBroker.Approval a))
          (@Eq Bool a.used true))
def ControlStack.TrustedBroker.Allowed : ControlStack.TrustedBroker.Roles →
  ControlStack.TrustedBroker.State →
    Nat →
      ControlStack.TrustedBroker.Request →
        ControlStack.TrustedBroker.Approval → ControlStack.TrustedBroker.Lease → Prop :=
fun roles s now r a l =>
  And (@Eq Bool s.halted false)
    (And (roles.agent r.uid)
      (And
        (@Eq (Option ControlStack.TrustedBroker.Approval) (s.approvals r.nonce)
          (@some ControlStack.TrustedBroker.Approval a))
        (And (@Eq Bool a.used false)
          (And (@Eq Nat a.digest r.digest)
            (And (@Eq Nat a.destination r.destination)
              (And (@Eq Nat a.agent r.uid)
                (And (@Eq Nat a.lease r.lease)
                  (And (roles.approver a.signer)
                    (And (@LT.lt Nat instLTNat now a.expires)
                      (And
                        (@Eq (Option ControlStack.TrustedBroker.Lease) (s.leases r.lease)
                          (@some ControlStack.TrustedBroker.Lease l))
                        (And (@Eq Nat l.agent r.uid)
                          (And (@Eq Bool l.revoked false)
                            (And (@LT.lt Nat instLTNat now l.expires)
                              (And
                                (@LT.lt Nat instLTNat (@OfNat.ofNat Nat (nat_lit 0) (instOfNatNat (nat_lit 0))) r.cost)
                                (And
                                  (@LE.le Nat instLENat
                                    (@HAdd.hAdd Nat Nat Nat (@instHAdd Nat instAddNat) l.used r.cost) l.budget)
                                  (And
                                    (@LE.le Nat instLENat
                                      (@HAdd.hAdd Nat Nat Nat (@instHAdd Nat instAddNat) s.spent r.cost) s.cap)
                                    (And (@Eq Bool (s.artifacts r.digest) true)
                                      (And (@Eq Bool (s.intact r.digest) true)
                                        (And
                                          (@Exists Nat fun reviewer =>
                                            @Eq (Option Nat) (s.reviews r.digest) (@some Nat reviewer))
                                          (Not
                                            (@Membership.mem Nat (List Nat) (@List.instMembership Nat) s.consumed
                                              r.nonce)))))))))))))))))))))
